  # 1) Firmware bauen  -> schreibt deployments/demo_ibex_uart_seg7/imem.mem + dmem.mem
  cd deployments/demo_ibex_uart_seg7/firmware
  cmake -B build -S . -DCMAKE_TOOLCHAIN_FILE=../../../toolchain/riscv-toolchain.cmake
  cmake --build build
  cd -

  # 2) Vivado-Aufrufe aus out/ starten -> Bitstream UND alle Vivado-Nebendateien
  #    (vivado.jou/.log, clockInfo.txt, .Xil/, ...) landen in out/.
  source ~/AMD/2025.2/Vivado/settings64.sh
  mkdir -p deployments/demo_ibex_uart_seg7/out && cd deployments/demo_ibex_uart_seg7/out

  #    Vollbuild -> out/demo_ibex_uart_seg7.{bit,mcs,mmi,dcp}
  vivado -mode batch -source ../scripts/build.tcl

  # 3a) Auf das Board per JTAG (fluechtig)
  vivado -mode batch -source ../scripts/program.tcl
  # 3b) ODER in den QSPI-Flash (JP1 auf JTAG, autonomer Boot)
  vivado -mode batch -source ../scripts/flash.tcl
  cd - >/dev/null

  # 4) Schneller Programm-Wechsel ohne Synthese (nur nach Schritt 1+2)
  cd deployments/demo_ibex_uart_seg7/out
  ../scripts/update_mem.sh
  vivado -mode batch -source ../scripts/program.tcl
  cd - >/dev/null

  # 5) UART mitlesen
  screen /dev/ttyUSB1 115200          # oder: stty -F /dev/ttyUSB1 115200 raw -echo -crtscts && cat /dev/ttyUSB1
